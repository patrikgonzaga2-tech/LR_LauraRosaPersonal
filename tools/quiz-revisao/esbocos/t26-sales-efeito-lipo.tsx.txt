type Oferta = { id: string; el: null | { preco: string; parc: string; dia: string }; sub: { preco: string; per: string; nota: string; parcCurta: string } }
const OFERTAS: Record<string, Oferta> = {
  A: { id: 'A', el: { preco: '37', parc: '12x de R$ 3,80', dia: '1,76' }, sub: { preco: '37', per: '/mês', nota: 'assinatura mensal · cancele quando quiser', parcCurta: 'R$ 37/mês' } },
  B: { id: 'B', el: { preco: '47', parc: '12x de R$ 4,83', dia: '2,24' }, sub: { preco: '37', per: '/mês', nota: 'assinatura mensal · cancele quando quiser', parcCurta: 'R$ 37/mês' } },
  C: { id: 'C', el: { preco: '37', parc: '12x de R$ 3,80', dia: '1,76' }, sub: { preco: '27', per: '/mês', nota: 'assinatura mensal · cancele quando quiser', parcCurta: 'R$ 27/mês' } },
  D: { id: 'D', el: { preco: '37', parc: '12x de R$ 3,80', dia: '1,76' }, sub: { preco: '97', per: '/3 meses', nota: 'plano trimestral · sai ≈ R$ 32 por mês', parcCurta: 'R$ 32/mês no trimestral' } },
  E: { id: 'E', el: null, sub: { preco: '37', per: '/mês', nota: 'assinatura mensal · cancele quando quiser', parcCurta: 'R$ 37/mês' } },
}
function readOferta(): Oferta {
  try { const o = new URLSearchParams(location.search).get('oferta'); if (o && OFERTAS[o]) return OFERTAS[o] } catch {}
  return OFERTAS.A
}
